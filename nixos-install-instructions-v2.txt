# NixOS Server Installation (No Encryption)
# ==========================================

# Connect to WiFi if needed (NixOS installer includes iwd)
# iwctl station wlan0 connect SSID

sudo su

# 1. Partition the drive
# ----------------------
# Using gdisk for GPT partitioning
# Adjust device name as needed (e.g., /dev/sda, /dev/nvme0n1)

gdisk /dev/nvme0n1
    # Create new GPT partition table:
    - o (confirm yes)

    # Partition 1: EFI Boot (2GiB)
    - n
    - (enter for default partition number: 1)
    - (enter for default first sector)
    - +2G
    - ef00 (EFI System)

    # Partition 2: Root (ext4) - adjust size as needed
    - n
    - (enter for default partition number: 2)
    - (enter for default first sector)
    - +100G  # Adjust size as needed for your use case
    - 8300 (Linux filesystem)

    # Partition 3: Home (btrfs) - remaining space
    - n
    - (enter for default partition number: 3)
    - (enter for default first sector)
    - (enter to use remaining space)
    - 8300 (Linux filesystem)

    # Write and exit
    - w

# 2. Format partitions
# --------------------
mkfs.fat -F32 -n NIXBOOT /dev/nvme0n1p1
mkfs.ext4 -L NIXROOT /dev/nvme0n1p2
mkfs.btrfs -L NIXHOME /dev/nvme0n1p3

# 3. Create btrfs @home subvolume
# -------------------------------
mount /dev/nvme0n1p3 /mnt
btrfs subvolume create /mnt/@home
umount /mnt

# 4. Mount filesystems
# --------------------
mount /dev/disk/by-label/NIXROOT /mnt

mkdir /mnt/boot
mount -o umask=0077 /dev/disk/by-label/NIXBOOT /mnt/boot

mkdir /mnt/home
mount -o subvol=@home,compress=zstd,noatime /dev/disk/by-label/NIXHOME /mnt/home

# 5. Clone configuration repository
# ----------------------------------
git clone git@github.com:dylankird/multi-server-nixos-config.git /mnt/etc/nixos
# Or use HTTPS if SSH keys are not available:
# git clone https://github.com/dylankird/multi-server-nixos-config.git /mnt/etc/nixos

# 6. Generate hardware configuration
# -----------------------------------
nixos-generate-config --root /mnt

# Move hardware config to host directory
mkdir -p /mnt/etc/nixos/hosts/<hostname>
mv /mnt/etc/nixos/hardware-configuration.nix /mnt/etc/nixos/hosts/<hostname>/

# Remove auto-generated configuration.nix (we use flake.nix)
rm /mnt/etc/nixos/configuration.nix

# 7. Review hardware configuration
# ---------------------------------
vim /mnt/etc/nixos/hosts/<hostname>/hardware-configuration.nix

# Consider adding if needed:
# hardware.enableAllFirmware = true;

# 8. Create or verify host config
# --------------------------------
# Ensure hosts/<hostname>/default.nix exists and imports hardware-configuration.nix
# Ensure flake.nix has an entry for this hostname

# 9. Install
# ----------
cd /mnt
sudo nixos-install --flake /mnt/etc/nixos#<hostname>

# If the installer complains about experimental features:
# sudo nixos-install --flake /mnt/etc/nixos#<hostname> --option experimental-features "nix-command flakes"

# 10. Reboot
# ----------
reboot
